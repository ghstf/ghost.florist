[taxonomies]

<div class="poem">
  <section>
    <p>A rat rotting in the walls;</p>
    <p>your smoke in its fur.</p>
  </section>

  <section>
    <p>Does it remember you?</p>
  </section>

  <section>
    <p>When the others find it,</p>
    <p>will they feed</p>
  </section>

  <section>
    <p class="indent">or know to leave it alone?</p>
  </section>
</div>
